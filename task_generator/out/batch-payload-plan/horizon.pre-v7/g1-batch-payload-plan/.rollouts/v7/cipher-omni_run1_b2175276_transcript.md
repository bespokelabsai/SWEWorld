def plan_batches(sizes, limits):
    if len(sizes) == 0:
        return []
    
    plans = []
    start_idx = 0
    count = 0
    bytes_sum = 0
    batch_index = 0
    
    for i, size in enumerate(sizes):
        if size > limits.max_bytes_per_batch:
            raise SingleRequestTooLargeError(row_idx=i, size_bytes=size, limit_bytes=limits.max_bytes_per_batch)
        
        new_count = count + 1
        new_bytes_sum = bytes_sum + size
        new_num_bytes = new_bytes_sum + (new_count - 1)
        
        exceeds = new_count > limits.max_requests_per_batch or new_num_bytes > limits.max_bytes_per_batch
        exact = new_count == limits.max_requests_per_batch or new_num_bytes == limits.max_bytes_per_batch
        
        if exceeds:
            # Close current batch without this request (count must be > 0 here)
            num_bytes = bytes_sum + max(0, count - 1)
            plans.append(PlannedBatch(index=batch_index, start_idx=start_idx, end_idx=i, num_requests=count, num_bytes=num_bytes))
            batch_index += 1
            # Start new batch with this request
            start_idx = i
            count = 1
            bytes_sum = size
        elif exact:
            # Close with this request
            plans.append(PlannedBatch(index=batch_index, start_idx=start_idx, end_idx=i+1, num_requests=new_count, num_bytes=new_num_bytes))
            batch_index += 1
            start_idx = i + 1
            count = 0
            bytes_sum = 0
        else:
            count = new_count
            bytes_sum = new_bytes_sum
    
    if count > 0:
        num_bytes = bytes_sum + max(0, count - 1)
        plans.append(PlannedBatch(index=batch_index, start_idx=start_idx, end_idx=len(sizes), num_requests=count, num_bytes=num_bytes))
    
    return plans
