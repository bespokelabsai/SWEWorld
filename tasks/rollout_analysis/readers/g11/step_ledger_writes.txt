@dataclass(frozen=True)
class StepPlan:
    """The shape of a training run, in one place."""

    step_unit: str
    num_examples: int
    batch_size: int
    epochs: int
    gradient_accumulation_steps: int
    batches_per_epoch: int
    total_batches: int
    total_steps: int
    trailing_window_batches: int
    dataset_signature: str

    def step_of_batch(self, batch_ordinal: int) -> int:
        """Return the 1-based optimizer step the 1-based ``batch_ordinal`` belongs to."""
        if batch_ordinal < 1:
            return 0
        return math.ceil(batch_ordinal / self.gradient_accumulation_steps)

    def is_step_boundary(self, batch_ordinal: int) -> bool:
        """Whether the optimizer steps after this 1-based batch ordinal."""
        if batch_ordinal < 1 or batch_ordinal > self.total_batches:
            return False
        if batch_ordinal == self.total_batches:
            return True
        return batch_ordinal % self.gradient_accumulation_steps == 0

    def epoch_of_batch(self, batch_ordinal: int) -> int:
        """Return the 1-based epoch the 1-based ``batch_ordinal`` falls in."""
        if batch_ordinal < 1:
            return 0
        return (batch_ordinal - 1) // self.batches_per_epoch + 1

    def batch_slice(self, batch_ordinal: int) -> Tuple[int, int]:
        """Return the ``(start, end)`` slice into the example list for a batch."""
        index_in_epoch = (batch_ordinal - 1) % self.batches_per_epoch
        start = index_in_epoch * self.batch_size
        end = min(start + self.batch_size, self.num_examples)
        return start, end

    def logging_steps(self, log_every_n_steps: int) -> Tuple[int, ...]:
        """Return the optimizer steps that are divisible by ``log_every_n_steps``."""
        if log_every_n_steps < 1:
            return ()
        return tuple(step for step in range(1, self.total_steps + 1) if step % log_every_n_steps == 0)

    def epoch_closing_steps(self) -> Tuple[int, ...]:
        """Return, per epoch, the step closing the window holding its last batch."""
        steps = []
        for epoch in range(1, self.epochs + 1):
            last_batch = min(epoch * self.batches_per_epoch, self.total_batches)
            steps.append(self.step_of_batch(last_batch))
        return tuple(sorted(set(steps)))

    def loss_history_steps(self, log_every_n_steps: int) -> Tuple[int, ...]:
        """Ascending, deduplicated union of logging steps and epoch-closing steps."""
        return tuple(sorted(set(self.logging_steps(log_every_n_steps)) | set(self.epoch_closing_steps())))

    def window_batches(self, step: int) -> Tuple[int, ...]:
        """Return the 1-based batch ordinals in the accumulation window of ``step``."""
        first = (step - 1) * self.gradient_accumulation_steps + 1
        last = min(step * self.gradient_accumulation_steps, self.total_batches)
        return tuple(range(first, last + 1))

    start_batch_in_epoch: int

    remaining_batches: int

def dataset_signature(examples: Sequence[Mapping[str, Any]]) -> str:
    """Fingerprint the content of every example, in order."""
    payload = json.dumps(
        list(examples),
        sort_keys=True,
        separators=(",", ":"),
        default=str,
        ensure_ascii=True,
    ).encode("utf-8")
    return f"{DATASET_SIGNATURE_PREFIX}-{len(examples)}-{xxhash.xxh64(payload).hexdigest()}"


def canonical_reasons(reasons: Iterable[str]) -> Tuple[str, ...]:
    """Return ``reasons`` deduplicated and ordered as ``CHECKPOINT_REASONS``.

    An empty input comes back empty, which is not an error; nor is repeating a
    reason. Anything outside :data:`CHECKPOINT_REASONS` raises
    :class:`StepLedgerError` right at the call site.
    """
    seen = set()
    for reason in reasons:
        if reason not in CHECKPOINT_REASONS:
            raise StepLedgerError(f"Unknown checkpoint reason {reason!r}; expected one of {CHECKPOINT_REASONS}.")
        seen.add(reason)
    return tuple(reason for reason in CHECKPOINT_REASONS if reason in seen)


def checkpoint_name(prefix: str, step: int) -> str:
    """Derive a checkpoint name deterministically from a prefix and a step."""
    return f"{prefix}-s{int(step):06d}"


def learning_rate_at(step: int, total_steps: int, base_lr: float, warmup_steps: int, *, min_lr_ratio: float = MIN_LR_RATIO) -> float:
    """Learning rate for optimizer ``step`` of a run of ``total_steps`` steps.

    Linear warmup peaking at ``base_lr`` on the last warmup step, then a linear
    decay across the steps that remain, landing on ``base_lr * min_lr_ratio``
    at ``total_steps``. The warmup is clipped to the run rather than raising,
    so a run shorter than its warmup still reaches ``base_lr`` at the end.
    """
    if total_steps <= 0:
        return float(base_lr)

    effective_warmup = min(max(int(warmup_steps), 0), int(total_steps))

    if effective_warmup > 0 and 1 <= step <= effective_warmup:
        return float(base_lr) * (step / effective_warmup)

    decay_steps = total_steps - effective_warmup
    if decay_steps <= 0:
        return float(base_lr)

    progress = (step - effective_warmup) / decay_steps
    progress = min(max(progress, 0.0), 1.0)
    return float(base_lr) * (min_lr_ratio + (1.0 - min_lr_ratio) * (1.0 - progress))

def plan_steps(
    num_examples: int,
    *,
    batch_size: int,
    epochs: int,
    gradient_accumulation_steps: int = 1,
    dataset_signature: str = "",
) -> StepPlan:
    """Plan a run in optimizer steps.

    The accumulation window is counted over the whole run and is not reset at
    an epoch boundary, so a window may span one. A short final window still
    counts as one whole optimizer step.
    """
    if num_examples == 0:
        raise StepLedgerError("Cannot plan a run over an empty dataset.")
    if num_examples < 0:
        raise StepLedgerError(f"num_examples must not be negative, got {num_examples}.")
    if batch_size < 1:
        raise StepLedgerError(f"batch_size must be at least 1, got {batch_size}.")
    if epochs < 1:
        raise StepLedgerError(f"epochs must be at least 1, got {epochs}.")
    if gradient_accumulation_steps < 1:
        raise StepLedgerError(f"gradient_accumulation_steps must be at least 1, got {gradient_accumulation_steps}.")

    batches_per_epoch = math.ceil(num_examples / batch_size)
    total_batches = batches_per_epoch * epochs
    total_steps = math.ceil(total_batches / gradient_accumulation_steps)
    trailing_window_batches = total_batches - (total_steps - 1) * gradient_accumulation_steps

    return StepPlan(
        step_unit=STEP_UNIT_OPTIMIZER,
        num_examples=num_examples,
        batch_size=batch_size,
        epochs=epochs,
        gradient_accumulation_steps=gradient_accumulation_steps,
        batches_per_epoch=batches_per_epoch,
        total_batches=total_batches,
        total_steps=total_steps,
        trailing_window_batches=trailing_window_batches,
        dataset_signature=dataset_signature,
    )


def plan_steps_for_config(config: "TinkerTrainerConfig", examples: Sequence[Mapping[str, Any]]) -> StepPlan:
    """Build a :class:`StepPlan` from a trainer config and the examples it will train on."""
    return plan_steps(
        len(examples),
        batch_size=config.batch_size,
        epochs=config.epochs,
        gradient_accumulation_steps=config.gradient_accumulation_steps,
        dataset_signature=dataset_signature(examples),
    )


def plan_packed_steps(num_examples: int, *, epochs: int, dataset_signature: str = "") -> StepPlan:
    """Plan a managed, server-side run: one packed step per epoch."""
    if num_examples == 0:
        raise StepLedgerError("Cannot plan a run over an empty dataset.")
    if num_examples < 0:
        raise StepLedgerError(f"num_examples must not be negative, got {num_examples}.")
    if epochs < 1:
        raise StepLedgerError(f"epochs must be at least 1, got {epochs}.")

    return StepPlan(
        step_unit=STEP_UNIT_PACKED,
        num_examples=num_examples,
        batch_size=num_examples,
        epochs=epochs,
        gradient_accumulation_steps=1,
        batches_per_epoch=1,
        total_batches=epochs,
        total_steps=epochs,
        trailing_window_batches=1,
        dataset_signature=dataset_signature,
    )

def plan_resume(plan: StepPlan, checkpoint: "CheckpointInfo") -> ResumePlan:
    """Validate a checkpoint against ``plan`` and say where the run picks back up.

    Validation runs in a fixed order: shape, then exhaustion, then consistency.
    A reshaped run is refused with :class:`ResumePlanMismatch` rather than
    warned about or silently restarted.
    """
    name = getattr(checkpoint, "name", "") or ""

    shape_fields = {
        "batch_size": (plan.batch_size, getattr(checkpoint, "batch_size", 0)),
        "dataset_signature": (plan.dataset_signature, getattr(checkpoint, "dataset_signature", "")),
        "gradient_accumulation_steps": (
            plan.gradient_accumulation_steps,
            getattr(checkpoint, "gradient_accumulation_steps", 0),
        ),
    }
    mismatched = tuple(sorted(field for field, (expected, found) in shape_fields.items() if expected != found))
    if mismatched:
        raise ResumePlanMismatch(
            checkpoint_name=name,
            mismatched_fields=mismatched,
            expected={field: shape_fields[field][0] for field in mismatched},
            found={field: shape_fields[field][1] for field in mismatched},
        )

    batches_completed = getattr(checkpoint, "batches_completed", 0)

    if batches_completed >= plan.total_batches:
        raise ResumePlanMismatch(
            checkpoint_name=name,
            mismatched_fields=("epochs",),
            expected={"epochs": plan.epochs, "total_batches": plan.total_batches},
            found={"batches_completed": batches_completed},
        )

    expected_step = plan.step_of_batch(batches_completed)
    if checkpoint.step != expected_step:
        raise ResumePlanMismatch(
            checkpoint_name=name,
            mismatched_fields=("step",),
            expected={"step": expected_step},
            found={"step": checkpoint.step},
        )

    return ResumePlan(
        checkpoint_name=name,
        start_batch_ordinal=batches_completed,
        start_epoch=batches_completed // plan.batches_per_epoch + 1,
        start_batch_in_epoch=batches_completed % plan.batches_per_epoch,
        completed_steps=checkpoint.step,
        remaining_batches=plan.total_batches - batches_completed,
    )
